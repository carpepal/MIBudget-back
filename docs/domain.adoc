== Domain Model

[plantuml]
....
rectangle Domain {
    class Usuario
    class Tablero 
    class Categoria
    class Presupuesto
    class Transaccion
    class Meta

    Usuario ---> Tablero : "tiene 1..*"
    Tablero ---> Meta : "tiene metas configuradas\n por el ahorro generado en\n el trablero" 
    Tablero ---> Transaccion : " almacena 0..*"
    Categoria ---> Transaccion : "cataloga 0..*"
    Categoria ---> Presupuesto : "configura"

}
....

== Aggregate Model
[plantuml]
....
rectangle Aggregate {
    class Usuario<<Aggregate>>
    class Tablero<<Aggregate Root>> 
    class Categoria<<Aggregate>>
    class Presupuesto
    class Transaccion<<Aggregate>>
    class Meta<<Aggregate>>


    Usuario ||--{ Tablero
    Tablero ||--{ Meta 
    Tablero ||--{ Transaccion 
    Categoria |o--{ Transaccion
    Categoria ||--o| Presupuesto
    }
....
